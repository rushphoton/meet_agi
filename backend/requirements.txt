# python-multipart: lets the settings screen upload documents. Without it: upload endpoint fails to load.
python-multipart==0.0.32
# httpx: HTTP client for the replay script and tests (and later vendor calls). Without it: urllib boilerplate.
httpx==0.28.1
# pytest: runs the tests. Without it: no automated proof.
pytest==9.1.1
